# How far examples/assets/p5.js gets through the engine. tests/p5_ratchet.cpp
# measures it; this file records it. The level may not go DOWN, and at the same
# level the blocker may not CHANGE, without the test failing.
#
# Only tools/p5-ratchet.py --advance writes this file. A test that edits its own
# expectations cannot fail, so advancing is a deliberate act.
#
# The ladder: 1 read, 2 lexed, 3 parsed, 4 compiled, 5 fits the bytecode,
# 6 top level ran, 7 defines p5. Rungs 8-12 (loads as a page, new p5(), setup,
# draw, golden) arrive with the work that makes them reachable.

# Wall-clock ceiling for lex + parse + compile + run, in milliseconds.
# Generous on purpose: it is here to catch the compiler's quadratic scans
# turning the first success into a forty-minute test, not to police noise.
# Ratchet it down as those are fixed.
budget-ms=120000

# TWO MODES, each with its own pawl.
#
# `level` is p5-min: the page defines `IS_MINIFIED`, which is what p5's own
# minified build defines. It collapses the Friendly Error System and an
# i18next initialize() that fetches a catalogue from cdn.jsdelivr.net.
#
# `full-level` is the SAME ladder with the flag left undefined, so the error
# system and the translator setup are both in play. Neither mode reaches the
# network - the fetch is guarded, and a test that reached out would fail for
# reasons that have nothing to do with this code.
#
# Both read 12. "p5 runs" is a different claim when half of p5 is switched
# off, which is why the second number is measured rather than assumed.
